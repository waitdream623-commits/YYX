## B

左右各一个相机，拍脸，脸朝向相机且身高严格大于才能被看到

可以对前k个翻转一次脸朝向，问最多拍到人数，

### 思路

- 记录从左右分别看，能否被照到(不考虑脸朝向)
- 先不考虑翻转记录能看到的
- 枚举翻转，记录最大位置k和人数(因为只能从开始连续一段)

```cpp
  int mx = 0;
    rep(i,1,n)
    {
        l[0][i] = (a[i] > mx); // 左侧可见
        mx = max(mx, a[i]);
    }

    mx = 0;
    for(int i = n; i >= 1; i--)
    {
        r[0][i] = (a[i] > mx); // 右侧可见
        mx = max(mx, a[i]);
    }

    int cur = 0;
    rep(i,1,n)//不翻转
    {
        if(s[i-1] == 'L')cur += l[0][i];
        else cur += r[0][i];
    }

    int ans = cur;
    int k = 0;

    rep(i,1,n)//枚举翻转位置，记录最大处
    {
        if(s[i-1] == 'L')
            cur += r[0][i] - l[0][i];//翻转后与翻转前变化
        else
            cur += l[0][i] - r[0][i];

        if(cur > ans)
        {
            ans = cur;
            k = i;
        }
    }

```

  const int BAD = 1'000'000'000;
  vector<int> f(n+1, BAD), pre(n+1, -1);
  vector<int> best(n+1, BAD), bestPre(n+1, -1);
  
  f[0] = 0;
  rep(i,1,n)
  {
      // i 当右端点，找指向同一格的最优左端点，没找到说明前面还没有合适左端点，不可能是1个点一段，至少3个，a[i]为正数
      int target = i-a[i];
      if(target >= 1 && best[target] != BAD)
      {
          f[i] = best[target]+1;
          pre[i] = bestPre[target];
      }
  
      // i 当左端点，登记给之后的右端点用
      target = i+a[i];
      if(target <= n && f[i-1] < best[target])
      {
          best[target] = f[i-1];
          bestPre[target] = i-1;
      }
  }
