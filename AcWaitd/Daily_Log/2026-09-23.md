class Solution {
public:

    using Matrix = vector<vector<long long>>;

    Matrix mul(const Matrix& a, const Matrix& b) {
        int n = a.size();
        int k = b.size();
        int m = b[0].size();

        Matrix c(n, vector<long long>(m, 0));

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                for (int x = 0; x < k; x++) {
                    c[i][j] += a[i][x] * b[x][j];
                }
            }
        }

        return c;
    }

    Matrix qpow(Matrix a, long long b) {
        int n = a.size();

        Matrix c(n, vector<long long>(n, 0));

        // 单位矩阵
        for (int i = 0; i < n; i++) {
            c[i][i] = 1;
        }

        while (b) {
            if (b & 1) {
                c = mul(c, a);
            }

            a = mul(a, a);
            b >>= 1;
        }

        return c;
    }

    int climbStairs(int n) {
        if (n <= 2) return n;

        Matrix A = {
            {1, 1},
            {1, 0}
        };

        Matrix res = qpow(A, n - 2);
        //f[n],f[n-1] = f[n-1],f[n-2] *A^1
        //最初就是{f[2]f[1]={2，1}*res
        return res[0][0] * 2 + res[0][1] * 1;
    }
};
