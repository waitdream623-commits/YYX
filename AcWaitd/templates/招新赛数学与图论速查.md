### GCD、快速幂、扩欧底板

[B3717](../Combinatorics/computing_combinations/B_3717_组合数问题.cpp) · [P5656](../Number_Theory/modular_arithmetic/P_5656_模板_二元一次不定方程_exgcd.cpp)

```cpp
// 非负数的 LCM：两者非零时，用 i128(a / gcd(a,b)) * b。
ll norm_mod(i128 a, ll p) {  // p > 0
    a %= p;
    if (a < 0) a += p;
    return ll(a);
}

ll qpow(ll a, ll b, ll p) {  // b >= 0
    ll ret = 1 % p;
    a = norm_mod(a,p);
    while (b) {
        if (b & 1) ret = ll(i128(ret)*a%p);
        a = ll(i128(a)*a%p);
        b >>= 1;
    }
    return ret;
}

ll exgcd(ll a, ll b, i128& x, i128& y) {  // a,b >= 0，且不同时为 0
    if (b == 0) { x = 1; y = 0; return a; }
    i128 x1,y1;
    ll g = exgcd(b,a%b,x1,y1);
    x = y1; y = x1-i128(a/b)*y1;
    return g;
}

// ax+by=c（a,b>0）：c%g 非零则无解；特解 x0,y0 乘 c/g。
// 正整数解的 k 范围：[ceil_div(1-x0,b/g), floor_div(y0-1,a/g)]。
i128 floor_div(i128 a,i128 b) { return a/b-(a%b<0); }  // b>0
i128 ceil_div(i128 a,i128 b) { return a/b+(a%b>0); }
```

vector<pair<ll,int>> factorize(ll x) {  // x >= 1
    vector<pair<ll,int>> res;
    for (ll j = 2; j <= x/j; ++j) {
        int c = 0;
        while (x%j == 0) x /= j, ++c;
        if (c) res.push_back({j,c});
    }
    if (x > 1) res.push_back({x,1});
    return res;
}

int p[N], cnt;
bool st[N];
void getprime(int n) {
    cnt = 0;
    fill(st,st+n+1,false);
    for (int i = 2; i <= n; ++i) {
        if (!st[i]) p[++cnt] = i;
        for (int j = 1; j <= cnt && p[j] <= n/i; ++j) {
            st[i*p[j]] = true;
            if (i%p[j] == 0) break;

    }
}

const int MOD = 998244353;  // 按题改
ll f[N], inv_f[N];

void init_comb(int n) {  // MOD 是质数；n < MOD；依赖 qpow
    f[0] = 1;
    for (int i = 1; i <= n; ++i) f[i] = f[i-1]*i%MOD;
    inv_f[n] = qpow(f[n],MOD-2,MOD);
    for (int i = n; i >= 1; --i) inv_f[i-1] = inv_f[i]*i%MOD;
}

ll C(int n,int m) {  // n 不超过预处理上限
    if (m < 0 || m > n) return 0;
    return f[n]*inv_f[m]%MOD*inv_f[n-m]%MOD;
}

### BFS

```cpp
vector<int> edge[N];
int dist[N], last[N];
void bfs(int n,int s) {
    fill(dist,dist+n+1,-1);  // -1 表示不可达
    fill(last,last+n+1,0);
    queue<int> q;
    dist[s] = 0; q.push(s);
    while (!q.empty()) {
        int x = q.front(); q.pop();
        for (int e : edge[x]) if (dist[e] == -1) {
            dist[e] = dist[x]+1; last[e] = x; q.push(e);
        }
    }
}
```

### Dijkstra

[P4779](../Graph_Theory/single_source_shortest/P_4779_模板_单源最短路径_标准版.cpp)

```cpp
const ll INF = 1LL<<62;  // 有限最短距离须小于 INF
vector<pair<int,ll>> edge[N];  // {终点, 非负边权}
ll dist[N];
bool st[N];

void dijk(int n,int s) {
    fill(dist,dist+n+1,INF);
    fill(st,st+n+1,false);
    using P = pair<ll,int>;  // {距离, 点}
    priority_queue<P,vector<P>,greater<P>> q;
    dist[s] = 0; q.push({0,s});
    while (!q.empty()) {
        auto [d,x] = q.top(); q.pop();
        if (st[x]) continue;
        st[x] = true;  // 出堆再标记
        for (auto [e,w] : edge[x]) {
            if (st[e] || w >= INF-d || d+w >= dist[e]) continue;
            dist[e] = d+w; q.push({dist[e],e});
        }
    }
}
```

### 并查集与 Kruskal

[P3367](../union-find_sets/P_3367_模板_并查集.cpp) · [Kruskal 原码](../Graph_Theory/minimum_spanning_tree/Kruskal_Algorithm.cpp)

```cpp
int fa[N], siz[N];

void init_dsu(int n) {
    for (int i = 1; i <= n; ++i) fa[i] = i, siz[i] = 1;
}

int find(int x) {
    return fa[x] == x ? x : fa[x] = find(fa[x]);
}

bool un(int x,int y) {
    int fx = find(x), fy = find(y);
    if (fx == fy) return false;
    if (siz[fx] > siz[fy]) swap(fx,fy);
    fa[fx] = fy; siz[fy] += siz[fx];
    return true;
}
```

```cpp
vector<array<ll,3>> edges;  // {权值, 起点, 终点}；每组读入前 clear
bool kruskal(int n,ll& ret) {  // 依赖 init_dsu、un；false 表示不连通
    init_dsu(n);
    sort(edges.begin(),edges.end());
    i128 sum = 0;
    ret = 0;
    int cnt = 0;
    for (auto [w,x,y] : edges) if (un(int(x),int(y))) {
        sum += w;
        if (++cnt == n-1) break;
    }
    if (cnt != n-1) return false;
    ret = ll(sum);  // 保证总权值能放入 ll
    return true;
}
```

### 拓扑排序

vector<int> edge[N], order;
int in[N];

bool topo(int n) {  // 有向图；false 表示有环
    fill(in,in+n+1,0);
    order.clear();
    for (int x = 1; x <= n; ++x)
        for (int e : edge[x]) ++in[e];
    queue<int> q;
    for (int x = 1; x <= n; ++x) if (!in[x]) q.push(x);
    while (!q.empty()) {
        int x = q.front(); q.pop();
        order.push_back(x);
        for (int e : edge[x]) if (--in[e] == 0) q.push(e);
    }
    return int(order.size()) == n;
}

int col[N];

bool bipartite(int n) {
    fill(col,col+n+1,-1);
    queue<int> q;
    for (int s = 1; s <= n; ++s) if (col[s] == -1) {
        col[s] = 0; q.push(s);
        while (!q.empty()) {
            int x = q.front(); q.pop();
            for (int e : edge[x]) {
                if (col[e] == col[x]) return false;

            }
        }
    }
    return true;

### 两遍 BFS 求直径

[CF2107D 原码](<../Contest/Codeforces Round 1023 (Div. 2)/D_Apple_Tree_Traversing.cpp>)

```cpp
vector<int> edge[N];
int dist[N], last[N];
bool vis[N];  // 点是否仍在树上；不删点就把 vis[1..n] 全设为 true

int bfs(int a) {  // a 必须仍在；仅用于无权树或森林的一个连通块
    queue<int> q;
    q.push(a); dist[a] = 0; last[a] = 0;
    int far = a;
    while (!q.empty()) {
        int x = q.front(); q.pop();
        if (dist[x] > dist[far] || (dist[x] == dist[far] && x > far)) far = x;
        for (int e : edge[x]) if (e != last[x] && vis[e]) {
            last[e] = x; dist[e] = dist[x]+1; q.push(e);
        }
    }
    return far;
}
// 不删点时：fill(vis+1,vis+n+1,true);
// int start = bfs(a), end = bfs(start);
// dist[end] 是边数，+1 是点数。按第二遍 last 从 end 回溯到 0。
// 找最远点时别扫未到达的点；多轮删点时别每轮清空 dist、last。
```

### LCA 与树上差分

[P3379 原码](../Tree/LCA/P_3379_模板_最近公共祖先_LCA.cpp)

```cpp
const int LIMIT = 19;  // n <= 5e5（N 也要开够）；更大时调整
vector<int> edge[N];
int deep[N], st[N][LIMIT+1], order[N], tot;
ll d[N];  // 差分中间值和 2*w 均须能放入 ll

void build(int n,int root) {  // 连通树；根的父亲是自己
    fill(d,d+n+1,0);
    deep[root] = 1;
    fill(st[root],st[root]+LIMIT+1,root);
    tot = 1; order[1] = root;
    for (int i = 1; i <= tot; ++i) {
        int x = order[i];
        for (int e : edge[x]) if (e != st[x][0]) {
            deep[e] = deep[x]+1; st[e][0] = x;
            for (int k = 1; k <= LIMIT; ++k) st[e][k] = st[st[e][k-1]][k-1];
            order[++tot] = e;
        }
    }
}

int query(int x,int y) {
    if (deep[x] < deep[y]) swap(x,y);
    for (int k = LIMIT; k >= 0; --k)
        if (deep[x]-(1LL<<k) >= deep[y]) x = st[x][k];
    if (x == y) return x;
    for (int k = LIMIT; k >= 0; --k)
        if (st[x][k] != st[y][k]) x = st[x][k], y = st[y][k];
    return st[x][0];
}

void add_nodes(int x,int y,ll w = 1) {
    int z = query(x,y);
    d[x] += w; d[y] += w; d[z] -= w;
    if (st[z][0] != z) d[st[z][0]] -= w;  // LCA 是根，只减一次
}

void add_edges(int x,int y,ll w = 1) {
    int z = query(x,y);
    d[x] += w; d[y] += w; d[z] -= 2*w;
}

void collect() {  // 批量修改后只调用一次；跳过根
    for (int i = tot; i >= 2; --i) {
        int x = order[i];
        d[st[x][0]] += d[x];
    }
}
// build(n,root) -> 逐条 add_nodes 或 add_edges（任选一种）-> collect()。
// 边答案取边的较深端点的 d。
```
